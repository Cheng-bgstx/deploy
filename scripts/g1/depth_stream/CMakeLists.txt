cmake_minimum_required(VERSION 3.18)
project(scalecar_depth LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(realsense2 REQUIRED)
find_package(ZLIB REQUIRED)
add_executable(scalecar-depth depth_stream.cpp)
target_link_libraries(scalecar-depth PRIVATE realsense2::realsense2 ZLIB::ZLIB)
install(TARGETS scalecar-depth RUNTIME DESTINATION bin)
enable_testing()
add_executable(depth-clock-tests test_depth_clock.cpp)
add_test(NAME depth_clock_jitter COMMAND depth-clock-tests)
